cmake_minimum_required(VERSION 3.21)
project(ticket_mvp LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)

set(TICKET_DESKTOP_TOOLS_DEFAULT ON)
if(ANDROID OR CMAKE_SYSTEM_NAME STREQUAL "iOS")
    set(TICKET_DESKTOP_TOOLS_DEFAULT OFF)
endif()
option(TICKET_BUILD_CLI "Build the console demonstration" ${TICKET_DESKTOP_TOOLS_DEFAULT})
option(BUILD_TESTING "Build core tests for the host" ${TICKET_DESKTOP_TOOLS_DEFAULT})
option(TICKET_BUILD_MOBILE "Build the Qt Quick Android/iOS application" ON)

add_library(ticket_core src/ticket_manager.cpp)
target_include_directories(ticket_core PUBLIC include)
target_link_libraries(ticket_core PUBLIC Threads::Threads)
set_target_properties(ticket_core PROPERTIES POSITION_INDEPENDENT_CODE ON)

set(ticket_targets ticket_core)
if(TICKET_BUILD_CLI)
    add_executable(ticket_demo src/main.cpp)
    target_link_libraries(ticket_demo PRIVATE ticket_core)
    list(APPEND ticket_targets ticket_demo)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(ticket_tests tests/ticket_manager_tests.cpp)
    target_link_libraries(ticket_tests PRIVATE ticket_core)
    add_test(NAME ticket_manager COMMAND ticket_tests)
    set_tests_properties(ticket_manager PROPERTIES TIMEOUT 30)
    list(APPEND ticket_targets ticket_tests)
endif()

foreach(target IN LISTS ticket_targets)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive-)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
endforeach()

if(TICKET_BUILD_MOBILE)
    add_subdirectory(mobile)
endif()
